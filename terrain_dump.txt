var history: Array = []

func _init(gw = 8, gh = 6, sub = 8):
	GW = gw
	GH = gh
	SUBDIV = sub
	RX = GW * SUBDIV + 1
	RZ = GH * SUBDIV + 1
	heights.resize(RX * RZ)
	heights.fill(0.0)
	_reset_weights(1)
func _reset_weights(m):
	weights = []
	for k in maxi(1, m):
		var a = []
		a.resize(RX * RZ)
		a.fill(0.0)
		weights.append(a)
	for v in RX * RZ:
		weights[0][v] = 1.0
func set_materials(defs):
	mats_def = defs
	_ensure_white()
	if weights.size() != maxi(1, defs.size()):
		_reset_weights(defs.size())
	_tex_cache = []
	_col_cache = []
	for k in 8:
		var d = defs[k] if k < defs.size() else {}
		var t = white_tex
		var en = 0.0
		var tp = str(d.get("tex", ""))
		if tp != "" and FileAccess.file_exists(tp):
			var im = Image.new()
			if im.load(tp) == OK:
				t = ImageTexture.create_from_image(im)
				en = 1.0
		_tex_cache.append(t)
		var col = d.get("col", Color(0.5, 0.5, 0.5))
		_col_cache.append(Color(col.r, col.g, col.b, en))
	_splat_img1 = null
	build()
	splat()
func _ensure_white():
	if white_tex == null:
		var wi = Image.create(1, 1, false, Image.FORMAT_RGBA8)
		wi.set_pixel(0, 0, Color(1, 1, 1))
		white_tex = ImageTexture.create_from_image(wi)
func reset_stroke():
	last_paint = null
func paint_at(x, z, radius, mi):
	if mi < 0 or mi >= weights.size():
		return
	var pts = []
	if last_paint != null:
		var d = Vector2(x - last_paint.x, z - last_paint.y)
		var L = d.length()
		var n = int(ceil(L / maxf(0.03, radius * 0.4)))
		for s in range(1, n + 1):
			pts.append(Vector2(last_paint.x + d.x * s / n, last_paint.y + d.y * s / n))
	else:
		pts.append(Vector2(x, z))
	last_paint = Vector2(x, z)
	for p in pts:
		_stamp(p.x, p.y, radius, mi)
	splat()
func _stamp(x, z, radius, mi):
	var ri = int(ceil(radius * SUBDIV))
	var ci = int(round(x * SUBDIV))
	var cj = int(round(z * SUBDIV))
	for j in range(maxi(0, cj - ri), mini(RZ, cj + ri + 1)):
		for i in range(maxi(0, ci - ri), mini(RX, ci + ri + 1)):
			var dx = float(i) / float(SUBDIV) - x
			var dz = float(j) / float(SUBDIV) - z
			var d = sqrt(dx * dx + dz * dz)
			if d > radius:
				continue
			var f = 1.0 - d / radius
			f = f * f * (3.0 - 2.0 * f)
			var v = i + j * RX
			for k in weights.size():
				if k == mi:
					weights[k][v] = lerp(weights[k][v], 1.0, f)
				else:
					weights[k][v] *= (1.0 - f)
			_norm(v)
	mark_rect_dirty(ci - ri, cj - ri, ci + ri, cj + ri)
func _norm(v):
	var s = 0.0
	for k in weights.size():
		s += weights[k][v]
	if s < 0.0001:
		for k in weights.size():
			weights[k][v] = 1.0 if k == 0 else 0.0
		return
	for k in weights.size():
		weights[k][v] /= s
func build():
	_need_build = true
func splat():
	_need_splat = true
func _process(_d):
	var now = Time.get_ticks_msec()
	if now - _last_build < 40:
		return
	if _need_build:
		_need_build = false
		_last_build = now
		_do_build_mesh()
	if _need_splat:
		_need_splat = false
		_last_build = now
		_do_splat()
func _ensure_shader():
	_ensure_white()
	if shader_mat == null:
		shader_mat = ShaderMaterial.new()
		var sh = Shader.new()
		sh.code = _sh_code()
		shader_mat.shader = sh
func _do_build_mesh():
	for c in get_children():
		c.free()
	_ensure_shader()
	var st = SurfaceTool.new()
	st.begin(Mesh.PRIMITIVE_TRIANGLES)
	for cy in GH:
		for cx in GW:
			if not has_cell(cx, cy):
				continue
			var i0 = cx * SUBDIV
			var j0 = cy * SUBDIV
			for sj in SUBDIV:
				for si in SUBDIV:
					var i = i0 + si
					var j = j0 + sj
					st.add_vertex(_v(i, j))
					st.add_vertex(_v(i + 1, j))
					st.add_vertex(_v(i + 1, j + 1))
					st.add_vertex(_v(i, j))
					st.add_vertex(_v(i + 1, j + 1))
					st.add_vertex(_v(i, j + 1))
	st.generate_normals()
	var am = st.commit()
	am.surface_set_material(0, shader_mat)
	var m = MeshInstance3D.new()
	m.mesh = am
	add_child(m)
	var st2 = SurfaceTool.new()
	st2.begin(Mesh.PRIMITIVE_LINES)
	for cy in GH:
		for cx in GW:
			if not has_cell(cx, cy):
				continue
			var i0 = cx * SUBDIV
			var j0 = cy * SUBDIV
			var i1 = i0 + SUBDIV
			var j1 = j0 + SUBDIV
			st2.add_vertex(_v(i0, j0))
			st2.add_vertex(_v(i1, j0))
			st2.add_vertex(_v(i1, j0))
			st2.add_vertex(_v(i1, j1))
			st2.add_vertex(_v(i1, j1))
			st2.add_vertex(_v(i0, j1))
			st2.add_vertex(_v(i0, j1))
			st2.add_vertex(_v(i0, j0))
	var am2 = st2.commit()
	var mi2 = StandardMaterial3D.new()
	mi2.albedo_color = Color(0.05, 0.05, 0.07)
	am2.surface_set_material(0, mi2)
	var m2 = MeshInstance3D.new()
	m2.mesh = am2
	m2.position.y = 0.02
	add_child(m2)
	if show_chunk_grid:
		var st3 = SurfaceTool.new()
		st3.begin(Mesh.PRIMITIVE_LINES)
		var seen = {}
		for key in chunk_mask:
			var p = key.split(",")
			var cx0 = int(p[0])
			var cy0 = int(p[1])
			for dd in [[0, 0], [1, 0], [-1, 0], [0, 1], [0, -1]]:
				var k2 = str(cx0 + dd[0]) + "," + str(cy0 + dd[1])
				if seen.has(k2):
					continue
				seen[k2] = 1
				var x0 = (cx0 + dd[0]) * CHUNK - OX
				var y0 = (cy0 + dd[1]) * CHUNK - OY
				var x1 = x0 + CHUNK
				var y1 = y0 + CHUNK
				var yy = 0.06
				st3.add_vertex(Vector3(x0, yy, y0))
				st3.add_vertex(Vector3(x1, yy, y0))
				st3.add_vertex(Vector3(x1, yy, y0))
				st3.add_vertex(Vector3(x1, yy, y1))
				st3.add_vertex(Vector3(x1, yy, y1))
				st3.add_vertex(Vector3(x0, yy, y1))
				st3.add_vertex(Vector3(x0, yy, y1))
				st3.add_vertex(Vector3(x0, yy, y0))
		var am3 = st3.commit()
		var mi3 = StandardMaterial3D.new()
		mi3.albedo_color = Color(0.9, 0.25, 0.2)
		mi3.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
		am3.surface_set_material(0, mi3)
		var m3 = MeshInstance3D.new()
		m3.mesh = am3
		add_child(m3)
func _do_splat():
	_ensure_shader()
	if _splat_img1 == null or _splat_img1.get_width() != RX or _splat_img1.get_height() != RZ:
		_init_splat_images()
	elif _dr.size() == 4:
		var i0 = _dr[0]
		var j0 = _dr[1]
		var i1 = _dr[2]
		var j1 = _dr[3]
		_dr = []
		for j in range(j0, j1 + 1):
			for i in range(i0, i1 + 1):
				var v = i + j * RX
				_splat_img1.set_pixel(i, j, Color(_w(0, v), _w(1, v), _w(2, v), _w(3, v)))
				_splat_img2.set_pixel(i, j, Color(_w(4, v), _w(5, v), _w(6, v), _w(7, v)))
		_splat_tex1.update(_splat_img1)
		_splat_tex2.update(_splat_img2)
	shader_mat.set_shader_parameter("splat1", _splat_tex1)
	shader_mat.set_shader_parameter("splat2", _splat_tex2)
	for k in 8:
		var t = _tex_cache[k] if k < _tex_cache.size() else white_tex
		var cc = _col_cache[k] if k < _col_cache.size() else Color(0.5, 0.5, 0.5, 0.0)
		shader_mat.set_shader_parameter("tex%d" % k, t)
		shader_mat.set_shader_parameter("ca%d" % k, cc)
	shader_mat.set_shader_parameter("uv_scale", UV_SCALE)
	shader_mat.set_shader_parameter("map_size", Vector2(GW, GH))
	shader_mat.set_shader_parameter("res", Vector2(RX, RZ))
func _sh_code():
	return "shader_type spatial;
uniform sampler2D splat1 : filter_linear;
uniform sampler2D splat2 : filter_linear;
uniform sampler2D tex0 : source_color, repeat_enable;
uniform sampler2D tex1 : source_color, repeat_enable;
uniform sampler2D tex2 : source_color, repeat_enable;
uniform sampler2D tex3 : source_color, repeat_enable;
uniform sampler2D tex4 : source_color, repeat_enable;
uniform sampler2D tex5 : source_color, repeat_enable;
uniform sampler2D tex6 : source_color, repeat_enable;
uniform sampler2D tex7 : source_color, repeat_enable;
uniform vec4 ca0;
uniform vec4 ca1;
uniform vec4 ca2;
uniform vec4 ca3;
uniform vec4 ca4;
uniform vec4 ca5;
uniform vec4 ca6;
uniform vec4 ca7;
uniform float uv_scale;
uniform vec2 map_size;
uniform vec2 res;
varying vec2 wxz;
varying vec2 suv;
void vertex() {
	vec4 wp = MODEL_MATRIX * vec4(VERTEX, 1.0);
	wxz = wp.xz;
	suv = (wp.xz / map_size * (res - vec2(1.0)) + vec2(0.5)) / res;
}
void fragment() {
	vec4 s1 = texture(splat1, suv);
	vec4 s2 = texture(splat2, suv);
	float w0 = s1.r; float w1 = s1.g; float w2 = s1.b; float w3 = s1.a;
	float w4 = s2.r; float w5 = s2.g; float w6 = s2.b; float w7 = s2.a;
	float ws = w0 + w1 + w2 + w3 + w4 + w5 + w6 + w7;
	w0 /= max(ws, 0.0001); w1 /= max(ws, 0.0001); w2 /= max(ws, 0.0001); w3 /= max(ws, 0.0001);
	w4 /= max(ws, 0.0001); w5 /= max(ws, 0.0001); w6 /= max(ws, 0.0001); w7 /= max(ws, 0.0001);
	vec2 uvt = wxz * uv_scale;
	vec3 c = vec3(0.0);
	c += mix(ca0.rgb, texture(tex0, uvt).rgb, ca0.a) * w0;
	c += mix(ca1.rgb, texture(tex1, uvt).rgb, ca1.a) * w1;
	c += mix(ca2.rgb, texture(tex2, uvt).rgb, ca2.a) * w2;
	c += mix(ca3.rgb, texture(tex3, uvt).rgb, ca3.a) * w3;
	c += mix(ca4.rgb, texture(tex4, uvt).rgb, ca4.a) * w4;
	c += mix(ca5.rgb, texture(tex5, uvt).rgb, ca5.a) * w5;
	c += mix(ca6.rgb, texture(tex6, uvt).rgb, ca6.a) * w6;
	c += mix(ca7.rgb, texture(tex7, uvt).rgb, ca7.a) * w7;
	ALBEDO = c;
	ROUGHNESS = 1.0;
	SPECULAR = 0.0;
}"
func _w(k, v):
	return weights[k][v] if k < weights.size() else 0.0
func _v(i, j):
	return Vector3(float(i) / float(SUBDIV), heights[i + j * RX], float(j) / float(SUBDIV))
func cell_h(gx, gy):
	var i = clampi(gx * SUBDIV + SUBDIV / 2, 0, RX - 1)
	var j = clampi(gy * SUBDIV + SUBDIV / 2, 0, RZ - 1)
	return heights[i + j * RX]
func sample_h(x, z):
	var fx = clampf(x, 0, GW) * SUBDIV
	var fz = clampf(z, 0, GH) * SUBDIV
	var i0 = int(fx)
	var j0 = int(fz)
	var i1 = mini(i0 + 1, RX - 1)
	var j1 = mini(j0 + 1, RZ - 1)
	var tx = fx - i0
	var tz = fz - j0
	var h00 = heights[i0 + j0 * RX]
	var h10 = heights[i1 + j0 * RX]
	var h01 = heights[i0 + j1 * RX]
	var h11 = heights[i1 + j1 * RX]
	return lerp(lerp(h00, h10, tx), lerp(h01, h11, tx), tz)
func can_move(fx, fy, tx, ty):
	return abs(cell_h(fx, fy) - cell_h(tx, ty)) <= MAX_STEP
func push_undo():
	var ws = []
	for k in weights:
		ws.append(k.duplicate())
	history.append({"h": heights.duplicate(), "w": ws})
	if history.size() > 8:
		history.pop_front()
func undo():
	if history.size() > 0:
		var sn = history.pop_back()
		heights = sn["h"]
		if sn["w"].size() == weights.size():
			weights = sn["w"]
		build()
		splat()
func apply_brush(x, z, radius, strength, mode):
	var ci = int(round(x * SUBDIV))
	var cj = int(round(z * SUBDIV))
	var ri = int(round(radius * SUBDIV))
	var aff = []
	for j in range(maxi(0, cj - ri), mini(RZ, cj + ri + 1)):
		for i in range(maxi(0, ci - ri), mini(RX, ci + ri + 1)):
			var d = sqrt((i - ci) * (i - ci) + (j - cj) * (j - cj)) / float(SUBDIV)
			if d > radius:
				continue
			aff.append([i, j, 1.0 - d / radius])
	if mode == "smooth":
		for a in aff:
			var i = a[0]
			var j = a[1]
			var s = 0.0
			var n = 0
			for dd in [[1,0],[-1,0],[0,1],[0,-1]]:
				var ii = i + dd[0]
				var jj = j + dd[1]
				if ii >= 0 and jj >= 0 and ii < RX and jj < RZ:
					s += heights[ii + jj * RX]
					n += 1
			if n > 0:
				heights[i + j * RX] = lerp(heights[i + j * RX], s / n, 0.5)
	else:
		for a in aff:
			var idx = a[0] + a[1] * RX
			if mode == "raise":
				heights[idx] += strength * a[2]
			elif mode == "lower":
				heights[idx] -= strength * a[2]
	build()
func to_data():
	return {"gw": GW, "gh": GH, "sub": SUBDIV, "h": heights, "w": weights, "chunks": chunk_mask, "ox": OX, "oy": OY, "migrated": migrated}
func _resample(old, oRX, oRZ):
	var osx = float(oRX - 1) / float(GW)
	var osz = float(oRZ - 1) / float(GH)
	var out = []
	out.resize(RX * RZ)
	for j in RZ:
		for i in RX:
			var x = float(i) / float(SUBDIV) * osx
			var z = float(j) / float(SUBDIV) * osz
			var i0 = int(x)
			var j0 = int(z)
			var i1 = mini(i0 + 1, oRX - 1)
			var j1 = mini(j0 + 1, oRZ - 1)
			var tx = x - i0
			var tz = z - j0
			out[i + j * RX] = lerp(lerp(old[i0 + j0 * oRX], old[i1 + j0 * oRX], tx), lerp(old[i0 + j1 * oRX], old[i1 + j1 * oRX], tx), tz)
	return out
func from_data(d):
	OX = int(d.get("ox", 0))
	OY = int(d.get("oy", 0))
	migrated = bool(d.get("migrated", false))
	var osub = int(d.get("sub", 4))
	var oRX = GW * osub + 1
	var oRZ = GH * osub + 1
	var h = d.get("h", [])
	if h.size() == RX * RZ:
		heights = h.duplicate()
	elif h.size() == oRX * oRZ:
		heights = _resample(h, oRX, oRZ)
	var w = d.get("w", [])
	if w.size() == weights.size() and w[0].size() == RX * RZ:
		weights = []
		for k in w:
			weights.append(k.duplicate())
	elif w.size() > 0 and w[0].size() == oRX * oRZ:
		weights = []
		for k in w:
			weights.append(_resample(k, oRX, oRZ))
		for v in RX * RZ:
			_norm(v)
	else:
		var m = d.get("mats", [])
		_reset_weights(mats_def.size())
		if m.size() == (oRX - 1) * (oRZ - 1):
			for j in RZ:
				for i in RX:
					var qi = mini(int(float(i) / SUBDIV * (oRX - 1)), oRX - 2)
					var qj = mini(int(float(j) / SUBDIV * (oRZ - 1)), oRZ - 2)
					var mi = int(m[qi + qj * (oRX - 1)])
					if mi < weights.size():
						for k in weights.size():
							weights[k][i + j * RX] = 1.0 if k == mi else 0.0
		elif m.size() == GW * GH:
			for j in RZ:
				for i in RX:
					var mi = int(m[clampi(i / SUBDIV, 0, GW - 1) + clampi(j / SUBDIV, 0, GH - 1) * GW])
					if mi < weights.size():
						for k in weights.size():
							weights[k][i + j * RX] = 1.0 if k == mi else 0.0
	var cm = d.get("chunks", {})
	if cm.size() == 0:
		var x0 = int(floor(float(OX) / float(CHUNK)))
		var y0 = int(floor(float(OY) / float(CHUNK)))
		var x1 = int(ceil(float(OX + GW) / float(CHUNK)))
		var y1 = int(ceil(float(OY + GH) / float(CHUNK)))
		for cy in range(y0, y1):
			for cx in range(x0, x1):
				cm[str(cx) + "," + str(cy)] = 1
	chunk_mask = cm
	for v in RX * RZ:
		var sum = 0.0
		for k in weights.size():
			sum += weights[k][v]
		if sum < 0.5 and weights.size() > 0:
			weights[0][v] = 1.0
	_splat_img1 = null
	build()
	splat()
func has_cell(x, y):
	var ax = x + OX
	var ay = y + OY
	if ax < 0 or ay < 0 or ax >= GW or ay >= GH:
		return false
	return chunk_mask.has(str(int(floor(float(ax) / float(CHUNK)))) + "," + str(int(floor(float(ay) / float(CHUNK)))))
func add_chunk(cx, cy):
	if cx < 0 or cy < 0:
		return {"ok": false, "dx": 0, "dy": 0}
	var key = str(cx) + "," + str(cy)
	if chunk_mask.has(key):
		return {"ok": false, "dx": 0, "dy": 0}
	chunk_mask[key] = 1
	if (cx + 1) * CHUNK > GW or (cy + 1) * CHUNK > GH:
		_resize(0, 0, maxi(GW, (cx + 1) * CHUNK), maxi(GH, (cy + 1) * CHUNK))
	else:
		build()
		splat()
	return {"ok": true, "dx": 0, "dy": 0}
func _resize(nox, noy, ngw, ngh):
	var sdx = OX - nox
	var sdy = OY - noy
	var oRX = RX
	var oRZ = RZ
	var oh = heights
	var ow = weights
	GW = ngw
	GH = ngh
	OX = nox
	OY = noy
	RX = GW * SUBDIV + 1
	RZ = GH * SUBDIV + 1
	heights = []
	heights.resize(RX * RZ)
	heights.fill(0.0)
	var nw = []
	for k in ow.size():
		var a = []
		a.resize(RX * RZ)
		a.fill(0.0)
		nw.append(a)
	for j in oRZ:
		for i in oRX:
			var ni = i + sdx * SUBDIV
			var nj = j + sdy * SUBDIV
			if ni >= 0 and nj >= 0 and ni < RX and nj < RZ:
				heights[ni + nj * RX] = oh[i + j * oRX]
				for k in ow.size():
					nw[k][ni + nj * RX] = ow[k][i + j * oRX]
	for v in RX * RZ:
		var sum = 0.0
		for k in nw.size():
			sum += nw[k][v]
		if sum < 0.5 and nw.size() > 0:
			nw[0][v] = 1.0
	weights = nw
	_splat_img1 = null
	build()
	splat()

func _init_splat_images():
	_splat_img1 = Image.create(RX, RZ, false, Image.FORMAT_RGBAF)
	_splat_img2 = Image.create(RX, RZ, false, Image.FORMAT_RGBAF)
	for j in RZ:
		for i in RX:
			var v = i + j * RX
			_splat_img1.set_pixel(i, j, Color(_w(0, v), _w(1, v), _w(2, v), _w(3, v)))
			_splat_img2.set_pixel(i, j, Color(_w(4, v), _w(5, v), _w(6, v), _w(7, v)))
	_splat_tex1 = ImageTexture.create_from_image(_splat_img1)
	_splat_tex2 = ImageTexture.create_from_image(_splat_img2)
	_dr = []
func mark_rect_dirty(i0, j0, i1, j1):
	i0 = clampi(i0, 0, RX - 1)
	i1 = clampi(i1, 0, RX - 1)
	j0 = clampi(j0, 0, RZ - 1)
	j1 = clampi(j1, 0, RZ - 1)
